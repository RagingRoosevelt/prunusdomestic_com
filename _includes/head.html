{%- comment -%} Section index pages are the only site.pages carrying a `section`, so this finds the parent index of any item page and inherits its title suffix. Page-level `title_suffix` wins; pages with no section fall to site-wide. {%- endcomment -%} {%- assign sect = site.pages | where: "section", page.section | first -%} {%- assign suffix = page.title_suffix | default: sect.title_suffix | default: site.title_suffix -%} {%- assign seo_title = page.seo_title | default: page.title -%} {%- assign full_title = seo_title | append: " | " | append: suffix -%} {%- assign desc = page.description | default: site.description -%} {%- assign og_desc = page.og_description | default: desc -%} {%- assign canonical = page.url | replace: "index.html", "" | absolute_url -%} {{ full_title }}
